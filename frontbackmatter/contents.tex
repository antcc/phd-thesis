    \chapter*{Symbols}

    \begingroup
    % Remove the colon just for this list
    \renewcommand{\descriptionlabel}[1]{\hspace*{\labelsep}\slsc{#1}}
    \begin{symbolsdesc}[labelwidth=6em, leftmargin=6em, style=nextline, itemsep=1pt]
        \item [\(v'\)] Transpose of a column vector \(v\)
        \item [\(\mathds{1}\)] Indicator function
        \item [\(\R^+\)] Positive real numbers in \((0, \infty)\)
        \item [\(\R^+_0\)] Non-negative real numbers in \([0, \infty)\)
        \item [\(L^1(\pp)\)] Space of integrable random elements with respect to a probability measure \(\pp\)
        \item [\(L^2(\pp)\)] Space of random elements with finite second moment with respect to a probability measure \(\pp\)
       \item[\(L^2[0,1]\)] Hilbert space of real square integrable functions on \([0,1]\)
        \item[\(\mathcal C[0,1]\)] Banach space of real continuous functions on \([0,1]\)
        \item [\(\Hcal(K)\)] Reproducing kernel Hilbert space associated with a covariance function \(K\)
        \item [\(\Hcal_0(K)\)] Pre-Hilbert space dense in \(\Hcal(K)\)
        \item [\(\|\cdot\|\)] Euclidean norm or generic matrix norm
        \item [\(\|\cdot\|_F\)] Frobenius norm on matrices
        \item [\(\|\cdot\|_{\max}\)] Maximum norm on matrices
        \item [\(\|\cdot\|_2\)] \(L^2[0,1]\)-norm or spectral norm on matrices
        \item [\(\|\cdot\|_{2,\pp}\)] Norm on \(L^2(\pp)\)
        \item [\(\|\cdot\|_\infty\)] Supremum norm on \(\mathcal C[0,1]\)
        \item [\(\|\cdot\|_K\)] Norm on \(\Hcal(K)\)
        \item [\(\dotprod{\cdot}{\cdot}_2\)] Inner product on \(L^2[0,1]\)
        \item [\(\dotprod{\cdot}{\cdot}_K\)] Inner product on \(\Hcal(K)\)
        \item [\(\Psi_X\)] Loève's isometry for the process \(X\)
        \item [\(\qq_X\)] Distribution of the process \(X\)
        \item [\(\Pi\)] Prior distribution
        \item [\(L(\cdot | \theta)\)] Likelihood function for a parameter \(\theta\)
        \item [\(\Pi_n\)] Posterior distribution based on \(n\) observations
        \item [\(\stackrel{\pp}{\longrightarrow}\)] Convergence in \(\pp\)-probability
        \item [\(a\lesssim b\)] There exists \(C>0\) such that \(a\leq Cb\)
        \item [\(O(\cdot)\)] Big-O notation
        \item [\(O_{\pp}(\cdot)\)] Big-O in \(\pp\)-probability notation (stochastic boundedness)
        \item [\(o(\cdot)\)] Little-o notation
        \item [\(\mathcal S\)] Discrete grid on \([0,1]\)
        \item [\(\mathfrak S_p\)] Permutations of \(\{1,\dots,p\}\)
        \item [\(\Theta\)] Parameter space
        \item [\(\mathfrak m\)] Lebesgue measure
        \item [\(\dkl(f \,\|\, g)\)] Kullback--Leibler divergence from \(f\) to \(g\)
        \item [\(V_2(f, g)\)] Second Kullback--Leibler variation from \(f\) to \(g\)
        \item [\(\kl(\Pi)\)] Kullback--Leibler support of a measure \(\Pi\)
        \item [\(d_H\)] Hellinger distance
        \item [\(N(\epsilon, \mathcal F, d)\)] \(\epsilon\)-covering number of the set \(\mathcal F\) with respect to the distance \(d\)
        \item [\(\mathcal U\)] Uniform distribution
        \item [\(\mathcal N(\mu, \sigma^2)\)] Normal distribution with mean \(\mu\) and variance \(\sigma^2\). When \(\sigma^2=0\) it is interpreted as a Dirac measure at \(\mu\).
        \item [\(\gp(\mu, K)\)] Gaussian process distribution with mean \(\mu\) and covariance \(K\)
        \item [\(\bff_{10}\)] Bayes factor in favor of the alternative hypothesis
        \item [\(\frac{d\pp}{d\qq}\)] Radon--Nikodym derivative of a measure \(\pp\) with respect to a reference measure \(\qq\)
    \end{symbolsdesc}
